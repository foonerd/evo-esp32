# evoconfig.txt - evo-esp32
#
# Copy this file to the root of the FAT32 microSD card that holds your
# music, edit the values, insert the card, power the player. The player
# reads it on every boot and writes evostatus.txt next to it saying what
# it applied. When you change a setting on the panel, the player writes
# it back into this file; your other lines and comments stay as they are.
#
# One "key = value" per line. Lines starting with # or ; are ignored.
# There are no comments after a value: everything after the first = is
# the value. Wrap a value in "double quotes" to keep leading or trailing
# spaces. An empty value puts the setting back to its default.

# The face's theme: evo-default, night-sky, sunrise, air or liquid.
theme = evo-default

# The face's style: disc (art in a circle, the progress on a ring round
# it) or card (art as a rounded square, the text over its foot).
style = disc

# Screen rotation in degrees clockwise: 0, 90, 180 or 270. Applied at
# boot.
rotation = 0

# The level at power-on, 0 to 100.
volume = 40

# The player's name on your network: letters, digits and hyphens, up to
# 30, no spaces. Leave empty for the default.
name = evo-player

# The Wi-Fi network to join, 2.4 or 5 GHz, WPA2 or WPA3. Leave the
# password empty for an open network. The address appears in Settings
# and in evostatus.txt once joined.
wifi_ssid = Your Network
wifi_password = your password

# The band to join it on when the network is on both: auto (5 GHz
# where the network is on it, either band where it is not), 5 (the
# same, said plainly) or 2.4. A NAS over 5 GHz answers in single
# milliseconds where 2.4 GHz takes tens, and this player read ten
# times as fast from the same server over 5 GHz; 5 GHz carries less
# far.
wifi_band = auto

# The order the queue plays in: on (shuffled) or off (as listed). And
# what happens at its end: all (starts again), off (stops), one (the
# same track again). Both are also a tap away in Settings.
shuffle = on
repeat = all

# What plays at power-on: a path on the card. Empty: the whole card, in
# the order above. A folder plays that folder; a playlist (.m3u, .pls)
# plays its first station, or the one named after a #, as in
#   start = Radio/Evo Selection.m3u#Classic FM
# A station waits for the network.
start =

# The Snapcast server to play from, as host or host:port (the port is
# 1704 when left out), for a multiroom served by snapserver. Empty: the
# player looks for one on the network. The server takes the sink while
# it streams and the card plays again when it stops.
snapserver =

# A network share the library lists under Network, beside the media
# servers it finds by itself. The protocol is in the address:
#   share = smb://nas.local/Music            SMB 2 or later (a NAS, a PC)
#   share = smb1://oldbox/music              SMB 1 only, for an old box
#   share = nfs://192.0.2.5/volume1/music     NFS (4.1, or 3 on an old box)
# A folder within the share may follow. The user and password are for
# SMB; leave the user empty for a guest share (an encrypted share needs
# an account); NFS has neither. A second share is share2, share2_user,
# share2_password, and so on up to share9.
share =
share_user =
share_password =

# The player's web page (http://<name>.local/ — the firmware update
# page) asks for this name and password when both are set; with either
# empty it is open to everyone on the network. The browser asks once;
# the password itself never crosses the network.
web_user =
web_password =

# The DAC board on the header, when the player cannot tell by itself
# (a board with no I2C part, such as a PCM5102A) or to override what it
# found; the names are on the wiki's Audio out page. And where the
# frames go when the board has a speaker and a DAC both: speaker, hat
# or both. Read now; acted on once the register of boards lands.
dac =
output = both
